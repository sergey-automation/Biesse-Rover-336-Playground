cmake_minimum_required(VERSION 3.13)
include($ENV{PICO_SDK_PATH}/external/pico_sdk_import.cmake)
project(sir88 C CXX ASM)
pico_sdk_init()
add_executable(sir88 main.c usb_descriptors.c)
pico_generate_pio_header(sir88 ${CMAKE_CURRENT_LIST_DIR}/uart9.pio)
target_include_directories(sir88 PRIVATE ${CMAKE_CURRENT_LIST_DIR})
target_link_libraries(sir88 pico_stdlib hardware_pio hardware_clocks hardware_irq tinyusb_device tinyusb_board)
pico_add_extra_outputs(sir88)
